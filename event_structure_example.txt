#  HOST_EVENTS_STRUCTURE #
##########################

# ATOMIC STRUCTURE (One Event Per JSON Object)
   {"time_of_packet":"hh:mm:ss:ms"},
# Example 1: Server Event
{
    "role": "POST_SERV",
    "timestamp_ref": "hh:mm:ss.ms",
    "host": {
        "id": "HEX",
        "os": "Linux/Windows",
        "ip": "...",
        "mac": "..."
    },
    "event": {
        "type": "event_type",
        "timestamp": "hh:mm:ss.ms",
        "details": "..."
    }
}

#######################
# EVENTS_TYPE/DETAILS #
#######################

# Note: In the atomic structure, these details appear inside the "event" object.

- event_type:
    - process_start:
        - timestamp: hh:mm:ss.ms
        - process:*
        - path:*
        - sha256:*
        - cmdline:*
        - parent:*
        - parent_path:
        - parent_sha_256:
        - user "administrator ,user ,guest":
        - cpu: (eg: 100%)
        - gpu: (eg: 100%)
        - ram: (eg: 100%)
        - disk: (eg: 100%)

    - process_kill:
        - timestamp: hh:mm:ss.ms
        - process:*
        - path:*
        - sha256:*
        - term_type:*
        - exit_code:*

    - file_create:
        - timestamp: hh:mm:ss.ms
        - file_type: (eg: ".exe, .txt")*
        - path:*
        - size:*
        - process:*
        - user:
        - owner:

    - file_modified:
        - timestamp: hh:mm:ss.ms
        - path:*
        - process:*
        - size_change:*
        - perm_change: (eg: "+w", "-r", "+x")*
        - user:
        - owner:

    - file_delete:
        - timestamp: hh:mm:ss.ms
        - path:*
        - process:*
        - size:*
        - user:
        - perm:*
        - owner:

    - service_create:
        - timestamp: hh:mm:ss.ms
        - name:*
        - path:*
        - start_type:*
        - account:
        - creator:

    - service_delete:
        - timestamp: hh:mm:ss.ms
        - name:*
        - path:*
        - account:
        - deleter:

    - service_modified:
        - timestamp: hh:mm:ss.ms
        - name:*
        - path:*
        - account:
        - user:

    - registry:
        - timestamp: hh:mm:ss.ms
        - op_type:*
        - reg_path:*
        - reg_val:*
        - reg_type:*
        - reg_user:
        - reg_owner:
        - reg_perm:*

    - network_connection:
        - timestamp: hh:mm:ss.ms
        - process:*
        - protocol:*
        - ip_local: true/false
        - dst_ip:
        - src_port:
        - dst_port:
        - sent:
        - recv:

    - dns_query:
        - timestamp: hh:mm:ss.ms
        - q_name: (eg: "google.com")*
        - q_type: (eg: ".com, .org")*
        - q_res: (eg: "ip")
        - q_port:
        - q_proto:*
